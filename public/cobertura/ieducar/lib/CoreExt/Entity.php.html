554    /**

557     * em que a classe está armazenada caso seja informado.
558     *
559     * Quando uma instância não é passada explicitamente, verifica-se se a
560     * instância já existe, retornado-a caso positivo e/ou instanciando uma nova

564     * armazenamento estático, pode ter efeitos indesejados ao ser usado por

579     * @param bool   $sticky   Se a instância da classe de ser "grundenda",

588        if (array_key_exists($search, self::$_classStorage) === true) {

590        } else {

600        }
